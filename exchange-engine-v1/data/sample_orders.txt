# =====================================================================
# v1 样例指令文件
# 运行：./build/file_driver data/sample_orders.txt
# 期望输出见 data/sample_expected.txt；逐步手推过程见 LEARNING_GUIDE.md 第 4 节
# 价格单位为 tick（例如 tick=0.01 元时，10020 表示 100.20 元）
# =====================================================================

# ---- 第 1 段：建立初始盘口（全部不交叉，只挂单）--------------------------
ADD 1 BUY 10000 20
ADD 2 BUY 10010 2
ADD 3 SELL 10020 8
ADD 5 BUY 10010 10
ADD 7 SELL 10030 5
ADD 9 SELL 10030 10
PRINT

# ---- 第 2 段：一笔买单吃多个价位（价格优先 + 同价位 FIFO）------------------
# 买 20 @10030：先吃最优卖价 10020（3 号 8 手），
# 再吃 10030：7 号先到，吃 5 手；9 号后到，吃剩下的 7 手
ADD 11 BUY 10030 20

# ---- 第 3 段：卖单部分成交，剩余挂单 --------------------------------------
# 卖 15 @10005：吃 10010 价位（2 号 2 手、5 号 10 手），
# 10000 低于限价 10005，停止；剩余 3 手挂在卖盘 10005
ADD 12 SELL 10005 15
PRINT

# ---- 第 4 段：撤单 --------------------------------------------------------
CANCEL 1
# 不存在的订单
CANCEL 99
# 已经全部成交的订单
CANCEL 3

# ---- 第 5 段：业务拒单 ----------------------------------------------------
# 9 号还挂着（剩 3 手）→ 重复 id
ADD 9 SELL 10040 1
# 数量非正
ADD 13 BUY 10000 0
# 价格非正
ADD 14 BUY -5 10

# ---- 第 6 段：maker 部分成交后保留队首位置 ---------------------------------
ADD 20 BUY 9990 5
ADD 21 BUY 9990 5
# 20 号被吃 3 手，剩 2 手，仍排在 21 号前面
ADD 22 SELL 9990 3
# 先吃 20 号剩下的 2 手，再吃 21 号 2 手
ADD 23 SELL 9990 4
PRINT
